const https = require('https'); function post(url, data) { return new Promise((resolve, reject) => { const body = new URLSearchParams(data).toString(); const urlObj = new URL(url); const options = { hostname: urlObj.hostname, path: urlObj.pathname, method: 'POST', headers: { 'Content-Type': 'application/x-www-form-urlencoded', 'Content-Length': Buffer.byteLength(body) } }; const req = https.request(options, (res) => { let data = ''; res.on('data', chunk => data += chunk); res.on('end', () => resolve(JSON.parse(data))); }); req.on('error', reject); req.write(body); req.end(); }); } exports.handler = async (event) => { const { code } = event.queryStringParameters || {}; if (!code) return { statusCode: 400, body: 'No code' }; const REDIRECT_URI = process.env.URL + '/api/gmail-callback'; try { const tokens = await post('https://oauth2.googleapis.com/token', { code, client_id: process.env.GOOGLE_CLIENT_ID, client_secret: process.env.GOOGLE_CLIENT_SECRET, redirect_uri: REDIRECT_URI, grant_type: 'authorization_code' }); // Guardar tokens en Supabase const sbUrl = process.env.SUPABASE_URL; const sbKey = process.env.SUPABASE_KEY; const saveRes = await fetch(`${sbUrl}/rest/v1/config`, { method: 'POST', headers: { 'apikey': sbKey, 'Authorization': `Bearer ${sbKey}`, 'Content-Type': 'application/json', 'Prefer': 'resolution=merge-duplicates,return=minimal' }, body: JSON.stringify({ key: 'gmail_tokens', value: JSON.stringify(tokens) }) }); return { statusCode: 302, headers: { Location: '/?gmail=connected' } }; } catch (e) { return { statusCode: 500, body: e.message }; } };
